1) Find the department-wise highest salary,but show only departments where the highest salary is greater than 55,000 = 

   SELECT DEPARTMENT , MAX(SALARY) AS HIGH_SALARY
   FROM employees
   group by DEPARTMENT
   HAVING max(SALARY) > 55000;

2) Find employees whose salary is greater than the salary of Rahul =
   
   SELECT EMP_NAME , SALARY FROM employees
   WHERE SALARY >
   (SELECT  SALARY FROM EMPLOYEES
   WHERE EMP_NAME = 'RAHUL') ;

3) Find employees who earn the same salary as Rahul =

   SELECT EMP_NAME , SALARY FROM employees
   WHERE SALARY =
   (SELECT  SALARY FROM EMPLOYEES
   WHERE EMP_NAME = 'RAHUL') ;

4) Find employees whose salary is greater than the average salary of the IT department = 

   SELECT EMP_NAME ,  SALARY
   FROM employees
   WHERE SALARY > (SELECT AVG(SALARY) FROM employees
                   WHERE DEPARTMENT = 'IT');

5) Find employees who earn the highest salary in the company =

   SELECT EMP_NAME ,  SALARY
   FROM employees
   WHERE SALARY = (SELECT MAX(SALARY) FROM employees);

6) Find employees who earn more than their department's average salary =

   SELECT EMP_NAME , DEPARTMENT , SALARY
   FROM employees E1
   WHERE  SALARY > (SELECT AVG(SALARY) FROM employees E2
                    WHERE E1.DEPARTMENT = E2.DEPARTMENT);
