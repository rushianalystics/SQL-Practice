1) Find the number of employees in each department =

   select department , count(*) as total_emp
   from employees
   group by department;

2) Find the average salary of employees in each department =

   select department , avg(salary) as avg_salary
   from employees
   group by department;

3) Find the highest salary in each department =

   select department , max(salary) as highest_salary
   from employees
   group by department;

4) Find the total salary paid in each department =

   select department , sum(salary) as total_salary
   from employees
   group by department;

5) Find the departments where the average salary is greater than 50,000 =

   select department , avg(salary) as avg_salary
   from employees
   group by department
   having avg(salary) > 50000;
