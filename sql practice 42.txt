-- 1) Find the average delivery rating for each weather condition.

select avg(delivery_rating) as avg_delivery_rating , weather_condition
from delivery_data
group by weather_condition;

-- 2)Find the average delivery time for each package type.

select avg(delivery_time_hours) as avg_delivery_time , package_type
from delivery_data
group by package_type;

-- 3)Find the number of delayed deliveries for each package type.

select count(*) as number_of_delayed_deliveries , package_type
from delivery_data
where `delayed` = 1
group by package_type;

-- 4)Find the average delivery cost for each delivery mode

select avg(delivery_cost) as avg_delivery_cost , delivery_mode
from delivery_data
group by delivery_mode;

-- 5)Find the average delivery cost for each delivery partner.

select avg(delivery_cost) as avg_delivery_cost , delivery_partner
from delivery_data
group by delivery_partner;
