1) SELECT ProductName, Price,
   CASE
   WHEN Price < 20 THEN 'Low Cost'
   WHEN Price BETWEEN 20 AND 50 THEN 'Medium Cost'
   ELSE 'High Cost'
   END AS PriceCategory
   FROM Products;

2) SELECT ProductID , Quantity ,
   CASE 
   WHEN Quantity > 50 THEN 'BEST'
   WHEN Quantity < 20 THEN 'BAD'
   ELSE 'AVG'
   END AS QTYLEVEL
   FROM OrderDetails;

3) SELECT EMP_NAME , SALARY , CITY ,
   CASE 
   WHEN SALARY > 60000 THEN 'BESTSALARY'
   WHEN SALARY < 40000 THEN 'BADSALARY'
   ELSE 'AVGSALARY'
   END AS SALAEYLEVEL
   FROM EMPLOYEE_SALES;

4) SELECT EMP_NAME , EXPERIENCE , 
   CASE 
   WHEN EXPERIENCE > 8 THEN 'EXCELLENTEMPLOYEE'
   WHEN EXPERIENCE < 3 THEN 'FRESHER'
   ELSE 'AVGEMPLOYEE'
   END AS EXPLEVEL
   FROM EMPLOYEE_SALES;
